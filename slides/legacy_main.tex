\section{Physics Motivation}
% ============================================================

\begin{frame}{Physics Motivation}
  \begin{itemize}
    \item \textbf{X(3872)}: exotic hadron candidate, nature still debated
      \begin{itemize}
        \item Compact tetraquark? $D^0\bar{D}^{*0}$ molecule? $\chi_{c1}(2P)$?
      \end{itemize}
    \item Study relative to conventional \psiP in the same decay channel:
      \begin{equation*}
        X(3872) / \psiP \to \PJGy \, \pi^{+}\pi^{-} \quad (\PJGy \to \mumu)
      \end{equation*}
    \item \textbf{pp reference} at \sqrtsnn = 5.36 TeV — baseline for nuclear modification
    \item \textbf{PbPb} data — probe QGP medium effects on exotic vs.\ conventional states
  \end{itemize}
\end{frame}

\section{Dataset \& Selection}
% ============================================================

\begin{frame}{Dataset \& Event Selection}
  \begin{block}{Data Samples}
    \begin{itemize}
      \item pp reference: 2024 pp data at \sqrtsnn = 5.36~TeV
      \item PbPb: 2023/2024 heavy-ion data (ongoing)
    \end{itemize}
  \end{block}

  \begin{block}{Reconstruction Channel}
    \begin{itemize}
      \item $X(3872) / \psiP \to \PJGy \, \pi^{+}\pi^{-}$, with $\PJGy \to \mumu$
      \item Full CMS detector: tracker + muon chambers
    \end{itemize}
  \end{block}

  \begin{block}{Pre-selection}
    \begin{itemize}
      \item Trigger: dimuon triggers (HLT)
      \item Muon ID: tight quality cuts
      \item Kinematic: \pt-dependent fiducial cuts
    \end{itemize}
  \end{block}
\end{frame}
