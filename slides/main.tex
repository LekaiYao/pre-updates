\begin{frame}{Can data replace an imperfect MC template?}
\begin{columns}[T]\column{.47\textwidth}\begin{block}{MC-template limitation}\begin{itemize}\item SPS modeling is not fully reliable\item Template choices propagate to the fraction\item A complementary data constraint is desirable\end{itemize}\end{block}
\column{.47\textwidth}\begin{exampleblock}{Exploratory goal}\begin{itemize}\item Construct DPS from mixed Data candidates\item Normalize in a DPS-enhanced region\item Derive SPS by subtraction\end{itemize}\end{exampleblock}\end{columns}
\vfill\centering\HL{\textbf{Explore whether a data-driven construction can replace the MC method when the simulation is not fully reliable.}}\vfill\atlasref
\end{frame}

\begin{frame}{SPS weight by subtraction}

\column{.44\textwidth}\textbf{Figure: data-driven SPS yields}\par\smallskip
\emph{Meaning: subtract normalized DPS from Data PP cell by cell.}
\[N_{\rm SPS}^i=N_{\rm PP}^i-N_{\rm DPS}^i.\]
\begin{alertblock}{Why can one control-region cell be negative?}Data and scaled mixed yields fluctuate independently. Their estimator difference is not constrained cell by cell.\end{alertblock}

\begin{frame}{Primary fraction: integral of the two-dimensional decomposition}

Test the pure-DPS control-region assumption by subtracting the MC-estimated SPS leakage before normalizing:

\end{block}
\column{.39\textwidth}
\begin{block}{Propagation}
Rebuild templates and weights, then rerun all four 1D component fits. The shift tests SPS-contamination sensitivity; it does not measure CR purity.
\end{block}

\begin{frame}{A pseudoexperiment propagates one possible statistical fluctuation}
\begin{columns}[T]\column{.47\textwidth}\begin{block}{One pseudoexperiment}Draw one correlated 14-cell Data-PP vector and one correlated 14-cell mixed-DPS vector from their covariance models.\end{block}
\begin{enumerate}\item Recompute CR normalization $\alpha$\item Recompute SPS/DPS cell yields and weights\item Recompute primary and reported fractions\end{enumerate}
\column{.47\textwidth}\begin{exampleblock}{Repeat 50,000 times}The standard deviation of the resulting estimator distribution defines the statistical uncertainty.\end{exampleblock}
\begin{itemize}\item Data: diagonal fit covariance\item Mixing: source-event bootstrap covariance\item Fixed: nominal 1D response coefficients\end{itemize}\end{columns}
\vfill\centering\textbf{The toys propagate the existing statistical information; they do not create new data.}
\end{frame}
