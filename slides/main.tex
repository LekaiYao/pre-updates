\begin{frame}{Definition of total acceptance times efficiency}
  The two reconstructed $J/\psi$ candidates and the event-level requirements
  enter one correction product:
  \[
    \epsilon_{\mathrm{total}} =
    \prod_{i=1}^{2}
    \left(
      A_{\eta} A_{p_{\mathrm{T}}}
      \epsilon_{\mathrm{RECO}}
      \epsilon_{\mathrm{ID}}
      \epsilon_{\mu\mu}
    \right)_{i}
    \epsilon_{\mathrm{HLT}}\epsilon_{\mathrm{evt}} .
  \]
  \vfill
  \begin{columns}[T]
    \column{0.47\textwidth}
      \textbf{Map coordinates}
      \[
        p_{\mathrm{T}}(J/\psi_{1}),\quad
        p_{\mathrm{T}}(J/\psi_{2})
        \in (10,40)\ \mathrm{GeV}.
      \]
    \column{0.47\textwidth}
      \textbf{Sample labels}
      \begin{itemize}
        \item SPS or DPS population
        \item No single-generator reinterpretation
      \end{itemize}
  \end{columns}
\end{frame}
